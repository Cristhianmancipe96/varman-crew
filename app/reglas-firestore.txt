    · CREAR y BORRAR únicamente SUS ventas (`local` = su correo) y solo con

    · Tocar en un producto SOLO el `stock`, y solo junto con una venta suya: las

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── ENDURECIDO 2026-07-12 (revisión de seguridad) ──
    // Antes bastaba "estar autenticado" para leer/escribir pedidos, inventario
    // y catálogo. Cualquiera que lograra crearse una cuenta en el proyecto
    // habría podido leer datos de clientes o dañar la tienda. Ahora TODO exige
    // ser del EQUIPO (lista de correos). Al contratar/cambiar vendedor: agregar
    // su correo aquí y volver a publicar las reglas.
    function esEquipo() {
      return request.auth != null
        && request.auth.token.email in [
          'c.mancipe.96@gmail.com',
          'andresvargasm91@gmail.com',
          'varmansneakersandclothes@gmail.com'   // vendedor (cuenta de la tienda) — agregado 2026-07-15
        ];
    }

    // LOCAL BÚNKER (2026-08-12): el local del socio es un libro APARTE del
    // negocio de VarMan Crew. Ojo: son datos de OTRO negocio (costos de
    // compra de 8 bodegas y lo que se les debe).
    // [2026-08-18] Se agregó el vendedor: con la pestaña Ventas tradicional
    // retirada, toda venta física de VarMan se registra aquí (bodega VARMAN),
    // así que necesita entrar. Esto le da acceso COMPLETO a Búnker, incluidas
    // las otras 7 bodegas del local — no solo lo de VarMan.
    // Si cambia, hay que actualizarlo en DOS lugares: aquí y SOCIOS_BUNKER
    // al inicio de app.jsx.
    function esBunker() {
      return request.auth != null
        && request.auth.token.email in [
          'andresvargasm91@gmail.com',
          'c.mancipe.96@gmail.com',
          'varmansneakersandclothes@gmail.com'
        ];
    }

    // Los dos socios: los ÚNICOS que pueden ver y tocar la caja
    function esSocio() {
      return request.auth != null
        && request.auth.token.email in [
          'c.mancipe.96@gmail.com',
          'andresvargasm91@gmail.com'
        ];
    }

    // ── LOCALES ALIADOS 2026-09-21 ──
    // Puntos de venta ajenos que exhiben pares de VarMan. Entran con su correo
    // y solo tocan lo suyo (ver arriba). Al dar acceso a un local nuevo: su
    // correo (en minúsculas) aquí Y en LOCALES al inicio de app.jsx, y republicar.
    function esLocal() {
      return request.auth != null
        && request.auth.token.email in [
          'larolastore1@gmail.com'   // La Rola Store, primer local aliado (2026-09-19)
        ];
    }

    // La fecha de HOY en Colombia como 'AAAA-MM-DD' (UTC-5, sin horario de
    // verano), calculada con el reloj del SERVIDOR: el local no puede registrar
    // ni borrar ventas de otro día cambiándole la fecha al celular.
    function hoyBogota() {
      let t = timestamp.value(request.time.toMillis() - 18000000);
      let m = t.month();
      let d = t.day();
      return string(t.year()) + '-' + (m < 10 ? '0' : '') + string(m)
                              + '-' + (d < 10 ? '0' : '') + string(d);
    }

    match /tiendas/{tienda} {
      // Documento de la tienda (guarda el logo). El local solo lo lee.
      allow read: if esEquipo() || esLocal();
      allow write: if esEquipo();

      // Un local solo mueve el stock de un producto JUNTO con una venta suya,
      // en la misma escritura atómica. `movLocal` es el id de esa venta:
      //  · venta NUEVA  (no existía y después sí)  → el stock baja lo vendido
      //  · venta BORRADA (existía y después no)    → el stock sube lo devuelto
      function movLocalValido(prodId) {
        let v = /databases/$(database)/documents/tiendas/$(tienda)/sales/$(request.resource.data.movLocal);
        return (
            !exists(v) && existsAfter(v)
            && getAfter(v).data.local == request.auth.token.email
            && getAfter(v).data.productoId == prodId
            && request.resource.data.stock == resource.data.stock - getAfter(v).data.cantidad
          ) || (
            exists(v) && !existsAfter(v)
            && get(v).data.local == request.auth.token.email
            && get(v).data.productoId == prodId
            && request.resource.data.stock == resource.data.stock + get(v).data.cantidad
          );
      }

      // Catálogo público de la página web (la página lee SIN login)
      match /catalogo/{doc} {
        allow read: if true;
        allow write: if esEquipo();
      }
      match /catalogoFotos/{doc} {
        allow read: if true;
        allow write: if esEquipo();
      }

      // CAJA DE LOS SOCIOS: solo los dos correos de arriba
      match /gastos/{doc} {
        allow read, write: if esSocio();
      }

      // Inventario y fotos: todo el equipo con login (incluye vendedor).
      // Un LOCAL aliado NO lee `products` (ahí viaja el costo): solo puede
      // cambiarle el `stock`, atado a una venta suya (movLocalValido) y sin
      // dejarlo nunca negativo. Las fotos sí las lee.
      match /products/{doc} {
        allow read, write: if esEquipo();
        allow update: if esLocal()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock', 'movLocal'])
          && request.resource.data.movLocal is string
          && resource.data.stock is int
          && request.resource.data.stock is int
          && request.resource.data.stock >= 0
          && movLocalValido(doc);
      }
      match /fotos/{doc} {
        allow read: if esEquipo() || esLocal();
        allow write: if esEquipo();
      }

      // VITRINA (2026-09-21): copia del inventario SIN costo para los locales
      // aliados. La escribe el equipo (la app la mantiene igual al inventario);
      // el local la lee y solo le mueve el `stock` al vender, para ver el
      // número nuevo al instante. Si la desordenara, solo se engaña a sí mismo:
      // el stock de verdad es el de `products` y la app del equipo la corrige.
      match /vitrina/{doc} {
        allow read: if esEquipo() || esLocal();
        allow write: if esEquipo();
        allow update: if esLocal()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock'])
          && request.resource.data.stock is int
          && request.resource.data.stock >= 0;
      }

      // VENTAS: todo el equipo registra/edita/borra ventas normales, pero
      // ANULAR (los campos anulada*) es solo de socios. Las reglas se suman
      // con "o": basta que una permita.
      match /sales/{doc} {
        allow read: if esEquipo();
        // socios: pueden todo (incluido anular y borrar anuladas)
        allow write: if esSocio();
        // el resto del equipo: solo escrituras SIN campos de anulación…
        allow create, update: if esEquipo()
          && !request.resource.data.keys().hasAny(
               ['anulada', 'anulada_fecha', 'anulada_por', 'anulada_motivo']);
        // …y solo puede borrar ventas que NO estén anuladas
        allow delete: if esEquipo() && !('anulada' in resource.data);

        // LOCAL aliado: solo SUS ventas (campo `local` = su correo). Para leer
        // la lista, la app consulta con where('local','==',correo); sin ese
        // filtro Firestore rechaza la consulta completa (así debe ser).
        allow read: if esLocal() && resource.data.local == request.auth.token.email;
        // Crear: suya, con fecha de HOY, sin costo ni campos de anulación (lista
        // cerrada de campos), y descontando el par en la misma escritura.
        allow create: if esLocal()
          && request.resource.data.local == request.auth.token.email
          && request.resource.data.id == doc
          && request.resource.data.fecha == hoyBogota()
          && request.resource.data.keys().hasOnly(
               ['id', 'productoId', 'fecha', 'cliente', 'modelo', 'talla', 'precio',
                'cantidad', 'canal', 'origen', 'local', 'localNombre'])
          && request.resource.data.productoId is string
          && request.resource.data.cantidad is int
          && request.resource.data.cantidad >= 1
          && request.resource.data.precio is number
          && request.resource.data.precio > 0
          && getAfter(/databases/$(database)/documents/tiendas/$(tienda)/products/$(request.resource.data.productoId)).data.movLocal == doc;
        // Borrar: suya, de HOY, que VarMan no haya anulado, y devolviendo el par
        // en la misma escritura. (No hay "editar": se borra y se registra de nuevo.)
        allow delete: if esLocal()
          && resource.data.local == request.auth.token.email
          && resource.data.fecha == hoyBogota()
          && !('anulada' in resource.data)
          && getAfter(/databases/$(database)/documents/tiendas/$(tienda)/products/$(resource.data.productoId)).data.movLocal == doc
          && getAfter(/databases/$(database)/documents/tiendas/$(tienda)/products/$(resource.data.productoId)).data.stock
             == get(/databases/$(database)/documents/tiendas/$(tienda)/products/$(resource.data.productoId)).data.stock + resource.data.cantidad;
      }

      // PEDIDOS DEL BOT DE WHATSAPP: todo el equipo con login los ve y los
      // gestiona (verificar pago, marcar enviado/entregado, cancelar, notas).
      // El bot los crea por su cuenta con la llave maestra (no usa esta regla).
      match /pedidos/{doc}  { allow read, write: if esEquipo(); }

      // Fotos de los comprobantes de pago (una por pedido, en colección aparte
      // para que la lista de pedidos no pese): el equipo solo las LEE; las
      // escribe el bot con la llave maestra.
      match /comprobantes/{doc} { allow read: if esEquipo(); }

      // ---- NUEVAS 2026-07-07 (v5, punto 5-bis y backlog 10-13) ----

      // Mapa catálogo↔inventario (PRIVADO: aquí van los códigos VRM y los
      // proveedores externos de cada ref). Lo edita el equipo desde la
      // pestaña Tienda; el bot lo LEE con la llave maestra.
      match /mapaCatalogo/{doc} { allow read, write: if esEquipo(); }

      // Nombres de bodegas externas reutilizables (los crea la pestaña Tienda)
      match /proveedores/{doc}  { allow read, write: if esEquipo(); }

      // Lista de espera de stock ("avísame cuando llegue la talla X"): la
      // escribe el bot; el equipo la ve en Pedidos y marca "ya avisé".
      match /listaEspera/{doc}  { allow read, write: if esEquipo(); }

      // Avisos que la app le deja al bot (reseña post-entrega, guía de envío).
      // La app solo CREA (y puede borrar un error); el bot los marca enviados
      // con la llave maestra.
      match /notificacionesPendientes/{doc} {
        allow read, create, delete: if esEquipo();
      }

      // ---- NUEVA 2026-07-09 ----

      // Pedidos ocultados (de prueba/duplicados): solo socios los ven y los
      // tocan. Ocultar/mostrar NO modifica el documento del pedido.
      match /pedidosOcultos/{doc} { allow read, write: if esSocio(); }

      // ---- NUEVA 2026-07-18 ----

      // Config del bot que se edita desde la app (pestaña Tienda): la
      // "referencia de la publicación" (refPauta) y las refs de la foto. El bot
      // la LEE con la llave maestra (no usa esta regla).
      // [SEGURIDAD 2026-07-21] Antes era `write: if esEquipo()`, lo que dejaba
      // a CUALQUIER cuenta del equipo (incluida la del vendedor) escribir
      // cualquier campo por API — por ejemplo `pausado` (apagar el bot de
      // ventas) o borrar el documento entero. Ahora la escritura libre es solo
      // de socios; el resto del equipo solo puede tocar las refs, nunca apagar
      // el bot. (La app ya escribía solo esos campos con merge; ahora la regla
      // lo IMPONE, no depende de la buena costumbre del cliente.)
      // ---- NUEVAS 2026-08-12: LOCAL BÚNKER ----
      // Las cuatro colecciones del local del socio (ventas del local, bodegas
      // que le despachan, pagos a esas bodegas y gastos del local). Nada de
      // esto toca el inventario ni la caja de VarMan Crew.
      match /bunkerVentas/{doc}      { allow read, write: if esBunker(); }
      match /bunkerProveedores/{doc} { allow read, write: if esBunker(); }
      match /bunkerPagos/{doc}       { allow read, write: if esBunker(); }
      match /bunkerGastos/{doc}      { allow read, write: if esBunker(); }

      // ---- NUEVA 2026-08-27: DÍA DE INVENTARIO ----
      // Cada documento es una sesión de conteo físico: qué se contó, quién lo
      // contó y el acta con las diferencias al cerrar. Es esEquipo() y no
      // esSocio() porque el que cuenta la bodega es el vendedor; el rastro de
      // quién cerró y qué ajustó queda dentro del documento.
      match /conteos/{doc} { allow read, write: if esEquipo(); }

      match /botConfig/{doc} {
        allow read:  if esEquipo();
        allow write: if esSocio();          // socios: control total (incl. pausar/crear/borrar)
        allow update: if esEquipo()         // vendedor: SOLO las refs, nunca `pausado`
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['refPauta','refPautaActualizado','refsFoto','refsFotoActualizado']);
      }
    }
  }
}
